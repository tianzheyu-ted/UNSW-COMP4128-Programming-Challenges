auto check = [&] (int k) -> bool {
        vector<Edge> edges;

        // 构造基础约束（相邻区间的前缀和约束）
        for (int i = 1; i < m; i++) {
            edges.push_back({i, i - 1, 0}); // S_{i-1} - S_i <= 0
            edges.push_back({i - 1, i, 1}); // S_i - S_{i-1} <= 1
        }

        // 构造线段的打击次数约束
        for (const auto& seg : comp_segments) {
            int L = seg.first;
            int R = seg.second;
            edges.push_back({R, L, -1}); // S_L - S_R <= -1
            edges.push_back({L, R, k});  // S_R - S_L <= k
        }

        // 3. Bellman-Ford 检测负权环
        // 初始化全部为 0，相当于隐式超级源点
        vector<int> dist(m, 0); 
        
        // 节点数为 m，最多进行 m 次大循环
        for (int i = 0; i < m; i++) {
            bool relaxed = false;
            for (const auto& edge : edges) {
                if (dist[edge.u] + edge.w < dist[edge.v]) {
                    dist[edge.v] = dist[edge.u] + edge.w;
                    relaxed = true;
                    
                    // 如果在第 m 次遍历仍能松弛，必定存在负权环
                    if (i == m - 1) {
                        return false; 
                    }
                }
            }
            // 剪枝：如果某一轮没有任何边被松弛，说明最短路已收敛，无负环
            if (!relaxed) {
                break;
            }
        }
        
        return true; // 没有负权环，不等式组有解
    };
