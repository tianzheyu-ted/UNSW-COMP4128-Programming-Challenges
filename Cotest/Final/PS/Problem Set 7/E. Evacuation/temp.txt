    // S -> (i, j, 0)
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if ('0' <= scientists[i][j] && scientists[i][j] <= '9') {
                network.add_edge(
                    n*n*(t+1) + rescue_capsules_map.size(), // 【修复1】把所有的 n*n*t 统一改成 n*n*(t+1)，避免和第 t 时刻的网格 ID 冲突
                    get_node(i, j, 0),
                    scientists[i][j] - '0'
                );
            }    
        }
    }
    
    for (int k = 1; k <= t; k++) {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (k - 1 >= visited[i][j]) continue;
                
                // (i, j, k-1) -> (i, j, k)
                if (k < visited[i][j]) { // 【修复2】加上对目的地时刻 k 的安全判定
                    network.add_edge(
                        get_node(i, j, k - 1),
                        get_node(i, j, k), 
                        network.INF
                    );
                }

                // (i, j, k-1) -> (i - 1, j, k)
                if (i - 1 >= 0 && (scientists[i - 1][j] != 'Z' && scientists[i - 1][j] != 'Y') && k < visited[i - 1][j]) { // 【修复2】末尾加上目的地安全判定
                    network.add_edge(
                        get_node(i, j, k - 1), 
                        get_node(i - 1, j, k), 
                        network.INF
                    );
                }

                // (i, j, k-1) -> (i + 1, j, k)
                if (i + 1 < n && (scientists[i + 1][j] != 'Z' && scientists[i + 1][j] != 'Y') && k < visited[i + 1][j]) { // 【修复2】末尾加上目的地安全判定
                    network.add_edge(
                        get_node(i, j, k - 1), 
                        get_node(i + 1, j, k), 
                        network.INF
                    );
                }

                // (i, j, k-1) -> (i, j - 1, k)
                if (j - 1 >= 0 && (scientists[i][j - 1] != 'Z' && scientists[i][j - 1] != 'Y') && k < visited[i][j - 1]) { // 【修复2】末尾加上目的地安全判定
                    network.add_edge(
                        get_node(i, j, k - 1), 
                        get_node(i, j - 1, k), 
                        network.INF
                    );
                }

                // (i, j, k-1) -> (i, j + 1, k)
                if (j + 1 < n && (scientists[i][j + 1] != 'Z' && scientists[i + 1][j + 1] != 'Y') && k < visited[i][j + 1]) { // 【修复2】末尾加上目的地安全判定
                    network.add_edge(
                        get_node(i, j, k - 1), 
                        get_node(i, j + 1, k), 
                        network.INF
                    );
                }

                if (rescue_capsules_map.count({i, j})) {
                    network.add_edge(
                        get_node(i, j, k - 1), 
                        n*n*(t+1) + rescue_capsules_map[{i, j}], // 【修复1】n*n*t -> n*n*(t+1)
                        network.INF
                    );
                    
                    // 【修复3】因为 k 循环到 t，这里顺手补上最后时刻 t 进入胶囊的机会
                    if (k == t && k < visited[i][j]) {
                        network.add_edge(
                            get_node(i, j, k), 
                            n*n*(t+1) + rescue_capsules_map[{i, j}], 
                            network.INF
                        );
                    }
                }
            }
        }
    }

    for (int i = 0; i < rescue_capsules_map.size(); i++) {
        network.add_edge(
            n*n*(t+1) + i,                                    // 【修复1】n*n*t -> n*n*(t+1)
            n*n*(t+1) + rescue_capsules_map.size() + 1,       // 【修复1】n*n*t -> n*n*(t+1)
            rescue_capsules[rescue_capsules_list[i].first][rescue_capsules_list[i].second] - '0'
        );
    }

    long long max_flow = network.dinic(
        n*n*(t+1) + rescue_capsules_map.size(),               // 【修复1】n*n*t -> n*n*(t+1)
        n*n*(t+1) + rescue_capsules_map.size() + 1            // 【修复1】n*n*t -> n*n*(t+1)
    );

    cout << max_flow << '\n';
